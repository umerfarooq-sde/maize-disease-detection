"""Independent saved-probability arithmetic; never performs image inference."""
import argparse
import csv
import hashlib
import json
from pathlib import Path
import sys
import numpy as np

ROOT = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(ROOT / "ml-training"))
from dataset_preparation import CLASS_LABELS, load_manifest
from fitness_integrity import verify_artifact_hashes
from model import read_checkpoint, build_model, restore_checkpoint
import torch

def read(path):
    return json.loads(path.read_text(encoding="utf-8"))

def hashfile(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def same(left, right):
    assert np.isclose(left, right, atol=2e-12, rtol=0), (left, right)

def auc(labels, scores):
    positive = scores[labels]
    negative = scores[~labels]
    differences = positive[:, None] - negative[None, :]
    return float(((differences > 0).sum() + 0.5 * (differences == 0).sum()) / differences.size)

def audit_csv(directory, manifest, split):
    rows = list(csv.DictReader((directory / "predictions.csv").open(encoding="utf-8", newline="")))
    membership = manifest.rows_for_split(split)
    assert [row["sample_id"] for row in rows] == [row.sample_id for row in membership]
    labels = np.array([int(row["true_index"]) for row in rows], dtype=np.int64)
    assert labels.tolist() == [row.class_index for row in membership]
    probs = np.array([[float(row[label]) for label in CLASS_LABELS] for row in rows])
    assert np.isfinite(probs).all() and (probs >= 0).all() and (probs <= 1).all()
    assert np.allclose(probs.sum(axis=1), 1, atol=2e-6, rtol=0)
    pred = probs.argmax(axis=1)
    assert pred.tolist() == [int(row["predicted_index"]) for row in rows]
    matrix = np.zeros((4, 4), dtype=np.int64)
    np.add.at(matrix, (labels, pred), 1)
    metrics = read(directory / "metrics.json")
    assert matrix.tolist() == metrics["confusion_matrix"]
    accuracy = float((labels == pred).mean())
    same(metrics["accuracy"], accuracy)
    per_class = []
    for k in range(4):
        tp = int(matrix[k, k]); fn = int(matrix[k].sum() - tp)
        fp = int(matrix[:, k].sum() - tp); tn = len(labels) - tp - fn - fp
        precision = tp / (tp + fp); recall = tp / (tp + fn)
        f1 = 2 * tp / (2 * tp + fp + fn); specificity = tn / (tn + fp)
        roc_auc = auc(labels == k, probs[:, k])
        actual = metrics["per_class"][k]
        assert actual["support"] == int(matrix[k].sum())
        for key, value in zip(("precision", "recall", "f1", "specificity", "roc_auc"), (precision, recall, f1, specificity, roc_auc)):
            same(actual[key], value)
        per_class.append(actual)
    for metric in ("precision", "recall", "f1"):
        same(metrics["macro_" + metric], float(np.mean([row[metric] for row in per_class])))
        same(metrics["weighted_" + metric], sum(row[metric] * row["support"] for row in per_class) / len(labels))
    same(metrics["roc_auc"]["macro"], float(np.mean([row["roc_auc"] for row in per_class])))
    same(metrics["roc_auc"]["weighted"], sum(row["roc_auc"] * row["support"] for row in per_class) / len(labels))
    return {"partition": split, "count": len(labels), "csv_sha256": hashfile(directory / "predictions.csv"), "metrics_sha256": hashfile(directory / "metrics.json"), "all_classification_and_auc_metrics_match": True, "accuracy": accuracy, "macro_f1": metrics["macro_f1"]}

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--experiment", type=Path, required=True)
    parser.add_argument("--manifest", type=Path, required=True)
    parser.add_argument("--test-output", type=Path)
    parser.add_argument("--output", type=Path, required=True)
    args = parser.parse_args()
    assert not args.output.exists()
    manifest = load_manifest(args.manifest)
    summary = read(args.experiment / "summary.json")
    assert summary["status"] == "completed" and summary["final_test_deferred"]
    assert summary["test_evaluated_once"] is False and summary["test_metrics"] is None
    assert not (args.experiment / "test").exists()
    assert summary["manifest_fingerprint"] == manifest.fingerprint
    checkpoint = args.experiment / summary["best_checkpoint"]
    assert hashfile(checkpoint) == summary["best_checkpoint_sha256"]
    environment = read(args.experiment / "environment.json")
    verify_artifact_hashes(ROOT, environment["source_hashes"])
    payload = read_checkpoint(checkpoint, class_names=CLASS_LABELS, preprocessing_hash=manifest.preprocessing_hash, manifest_hash=manifest.fingerprint)
    settings = read(args.experiment / "hyperparameters.json")
    assert payload["hyperparameters"] == settings
    assert payload["seed"] == settings["seed"] == manifest.split_seed == 20261007
    assert payload["dataset_version"] == manifest.version
    history = read(args.experiment / "history.json")
    best = max(history, key=lambda row: (row["validation_macro_f1"], -row["validation_loss"]))
    assert best["epoch"] == payload["epoch"] == summary["best_epoch"]
    assert len(history) == summary["epochs_completed"]
    assert [row["epoch"] for row in history] == list(range(1, len(history) + 1))
    model = build_model(4, pretrained=False)
    torch.set_num_threads(2)
    restore_checkpoint(model, payload, 4)
    with torch.inference_mode():
        logits = model(torch.zeros(1, 3, 224, 224))
    assert tuple(logits.shape) == (1, 4) and torch.isfinite(logits).all()
    evidence = [audit_csv(args.experiment / split, manifest, split) for split in ("train", "validation")]
    if args.test_output:
        final_summary = read(args.test_output / "summary.json")
        assert final_summary["test_evaluated_once"] is True
        evidence.append(audit_csv(args.test_output / "raw", manifest, "test"))
    result = {"passed": True, "experiment_id": summary["experiment_id"], "manifest_fingerprint": manifest.fingerprint, "checkpoint_sha256": hashfile(checkpoint), "source_hashes_match": True, "best_epoch_independently_selected": payload["epoch"], "parameter_count": sum(p.numel() for p in model.parameters()), "zero_input_finite_1x4": True, "source_images_read": False, "audits": evidence}
    args.output.parent.mkdir(parents=True, exist_ok=True)
    with args.output.open("x", encoding="utf-8") as stream:
        json.dump(result, stream, indent=2, allow_nan=False); stream.write("\n")
    print(json.dumps(result, allow_nan=False))

if __name__ == "__main__":
    main()
