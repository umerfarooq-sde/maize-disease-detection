def check(condition, message):
    if not condition:
        raise ValueError(message)


def read(path):
    def invalid_constant(value):
        raise ValueError("Non-finite JSON value.")
    return json.loads(path.read_bytes(), parse_constant=invalid_constant)


def sha(data):
    return hashlib.sha256(data).hexdigest()


def relative(path):
    resolved = path.resolve(strict=True)
    check(resolved.is_relative_to(ROOT), "Artifact escapes repository.")
    return resolved.relative_to(ROOT).as_posix()


def reference(path):
    return {"path": relative(path), "sha256": file_digest(path), "size_bytes": path.stat().st_size}


def bind(actual, expected, description):
    for key, value in expected.items():
        check(key in actual and actual[key] == value, description + " differs: " + key)
        if isinstance(value, bool):
            check(actual[key] is value, description + " boolean differs: " + key)


def write_json(path, payload):
    with path.open("x", encoding="utf-8", newline="\n") as stream:
        json.dump(payload, stream, indent=2, ensure_ascii=False, allow_nan=False)
        stream.write("\n")


def metric_audit(folder, manifest, partition):
    with (folder / "predictions.csv").open(encoding="utf-8", newline="") as stream:
        reader = csv.DictReader(stream)
        check(reader.fieldnames == ["sample_id", "true_index", "predicted_index", *CLASS_LABELS], "CSV schema differs.")
        records = list(reader)
    membership = manifest.rows_for_split(partition)
    check([row["sample_id"] for row in records] == [row.sample_id for row in membership], "Exact split membership differs.")
    labels = np.array([int(row["true_index"]) for row in records], dtype=np.int64)
    probabilities = np.array([[float(row[label]) for label in CLASS_LABELS] for row in records])
    check(labels.tolist() == [row.class_index for row in membership], "CSV labels differ from index.")
    check(np.isfinite(probabilities).all(), "CSV probabilities are non-finite.")
    check(probabilities.argmax(axis=1).tolist() == [int(row["predicted_index"]) for row in records], "CSV predictions differ from argmax.")
    calculated = dict(compute_metrics(labels, probabilities, CLASS_LABELS))
    metrics = read(folder / "metrics.json")
    bind(metrics, calculated, partition + " saved arithmetic")
    return calculated, {"partition": partition, "sample_count": len(records), "csv_sha256": file_digest(folder / "predictions.csv"), "metrics_sha256": file_digest(folder / "metrics.json"), "membership_labels_argmax_and_metrics_match": True}


def privacy_check(name, data):
    if name.endswith(".png"):
        check(data.startswith(b"\x89PNG\r\n\x1a\n"), "Numerical plot has invalid PNG signature.")
        return
    text = data.decode("utf-8")
    check(str(PRIVATE_ROOT).casefold() not in text.casefold(), "Private dataset path detected.")
    check(PRIVATE_ROOT.as_posix().casefold() not in text.casefold(), "Private dataset path detected.")
    check(not re.search(r"(?<![A-Za-z0-9])[A-Za-z]:[/\\]", text), "Absolute Windows path detected.")
    check(not re.search(r"(?:data:image/[A-Za-z0-9.+-]+;base64,|-----BEGIN (?:RSA |EC |OPENSSH )?PRIVATE KEY-----)", text), "Binary image payload or private key detected.")
    if name.endswith(".json"):
        payload = json.loads(text)
        def inspect(value):
            if isinstance(value, dict):
                for key, child in value.items():
                    check(key.lower() not in {"gpslatitude", "gpslongitude", "gpsinfo", "image_base64", "image_bytes", "pixel_data", "raw_pixels"}, "Forbidden image/private metadata detected.")
                    inspect(child)
            elif isinstance(value, list):
                for child in value:
                    inspect(child)
        inspect(payload)
